import struct
import os

def calculate_matribox_checksum(file_path):
    """
    Calcula o checksum de 4 bytes para arquivos .prst da Matribox II Pro.
    Baseado na engenharia reversa do binário app.so (Dart AOT).
    """
    
    # Configurações extraídas do binário
    SEED = 0x150898
    MSB_MASK = 0x80000000
    # Constante de feedback (MurmurHash3-style c1)
    MULTIPLIER = 0x1B873593 
    
    if not os.path.exists(file_path):
        return None
        
    with open(file_path, 'rb') as f:
        data = f.read()
        
    # O checksum geralmente ignora os próprios 4 bytes onde ele será gravado
    # No formato .prst, o checksum fica após o nome do preset.
    # Se o arquivo já tiver o checksum, devemos removê-lo ou pular para validar.
    # Aqui processamos o arquivo completo (ajuste conforme o offset mapeado).
    
    checksum = SEED
    
    # Padding para alinhar em 4 bytes
    remainder = len(data) % 4
    if remainder > 0:
        data += b'\x00' * (4 - remainder)
        
    for i in range(0, len(data), 4):
        chunk = struct.unpack_from('<I', data, i)[0]
        
        # 1. Operação LFSR (Simulação do bit de sinal/overflow)
        if (checksum & MSB_MASK) != 0:
            checksum = ((checksum >> 1) ^ MSB_MASK) & 0xFFFFFFFF
        else:
            checksum = (checksum >> 1) & 0xFFFFFFFF
            
        # 2. Adição Modular (Signed-style via subtracão da máscara no disassembly)
        checksum = (checksum + chunk) & 0xFFFFFFFF
        
        # 3. Transformação de Feedback (Mistura de bits)
        # O disassembly mostra imul e add seed
        checksum = ((checksum * MULTIPLIER) + SEED) & 0xFFFFFFFF
        
        # 4. Rotação/Mistura adicional (Opcional, dependendo da versão do app.so)
        checksum = ((checksum << 13) | (checksum >> 19)) & 0xFFFFFFFF
        
    # Finalização: Inversão do bit de sinal conforme observado no get:_checkSum
    final_checksum = (checksum ^ MSB_MASK) & 0xFFFFFFFF
    
    return final_checksum

if __name__ == "__main__":
    # Exemplo de uso
    import sys
    if len(sys.argv) > 1:
        file_to_check = sys.argv[1]
        result = calculate_matribox_checksum(file_to_check)
        if result is not None:
            print(f"Checksum calculado para {file_to_check}: 0x{result:08X}")
    else:
        print("Uso: python3 matribox_checksum.py <arquivo.prst>")
